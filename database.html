🗄️ High-Speed Railway Database (MySQL / PostgreSQL)

Database Name: railway_ticket_db

Relational DDL Scripts

Module 5 defines the backend relational database structure required to power the high-speed bullet train ticket reservation system. Below are the production-grade SQL table creation scripts (DDL) with primary keys, foreign key constraints, indexes, and sample data.

👤 1. USERS

Stores passenger accounts, encrypted passwords, emails, and contact numbers.

🚉 2. STATIONS

Stores railway station codes (NDLS, MMCT, MAS), city names, and state details.

🚄 3. TRAINS

Stores high-speed bullet train models, numbers, maximum speed, and types.

🎫 4. BOOKINGS

Stores 10-digit PNR, user link, total fare paid, and booking status (CNF/WL).

👥 5. PASSENGERS

Stores individual passenger names, ages, genders, and allocated coach/berths.

💳 6. PAYMENTS

Stores transaction IDs, payment gateways (Card/UPI/NetBanking), and timestamps.

📌 Database Initialization 
-- 1. Create and switch to railway reservation database
CREATE DATABASE IF NOT EXISTS railway_ticket_db 
CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

USE railway_ticket_db;
👤 Table 1: USERS 
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(120) NOT NULL UNIQUE,
    phone_number VARCHAR(15) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('passenger', 'admin') DEFAULT 'passenger',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
🚉 Table 2: STATIONS 
CREATE TABLE stations (
    station_id INT AUTO_INCREMENT PRIMARY KEY,
    station_code VARCHAR(10) NOT NULL UNIQUE,
    station_name VARCHAR(100) NOT NULL,
    state VARCHAR(50) NOT NULL
);
🚄 Table 3: TRAINS 
CREATE TABLE trains (
    train_id INT AUTO_INCREMENT PRIMARY KEY,
    train_number VARCHAR(10) NOT NULL UNIQUE,
    train_name VARCHAR(100) NOT NULL,
    train_type VARCHAR(50) NOT NULL,
    speed_kmh VARCHAR(20) NOT NULL,
    origin_station_id INT NOT NULL,
    dest_station_id INT NOT NULL,
    departure_time TIME NOT NULL,
    arrival_time TIME NOT NULL,
    duration VARCHAR(20) NOT NULL,
    punctuality_rate VARCHAR(10) DEFAULT '99.0%',
    FOREIGN KEY (origin_station_id) REFERENCES stations(station_id),
    FOREIGN KEY (dest_station_id) REFERENCES stations(station_id)
);
🎫 Table 4: BOOKINGS 
CREATE TABLE bookings (
    booking_id INT AUTO_INCREMENT PRIMARY KEY,
    pnr_number VARCHAR(20) NOT NULL UNIQUE,
    user_id INT NOT NULL,
    train_id INT NOT NULL,
    journey_date DATE NOT NULL,
    class_code VARCHAR(10) NOT NULL,
    quota VARCHAR(30) DEFAULT 'General',
    total_fare DECIMAL(10,2) NOT NULL,
    booking_status ENUM('CONFIRMED', 'WAITLISTED', 'CANCELLED') DEFAULT 'CONFIRMED',
    booking_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (train_id) REFERENCES trains(train_id)
);
👥 Table 5: PASSENGERS 
CREATE TABLE passengers (
    passenger_id INT AUTO_INCREMENT PRIMARY KEY,
    booking_id INT NOT NULL,
    passenger_name VARCHAR(100) NOT NULL,
    age INT NOT NULL,
    gender ENUM('Male', 'Female', 'Transgender') NOT NULL,
    coach_number VARCHAR(10) NOT NULL,
    seat_number INT NOT NULL,
    berth_preference VARCHAR(30),
    status VARCHAR(20) DEFAULT 'CNF',
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id) ON DELETE CASCADE
);
💳 Table 6: PAYMENTS 
CREATE TABLE payments (
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    booking_id INT NOT NULL,
    transaction_ref VARCHAR(50) NOT NULL UNIQUE,
    payment_method ENUM('CARD', 'UPI', 'NETBANKING') NOT NULL,
    amount_paid DECIMAL(10,2) NOT NULL,
    payment_status ENUM('SUCCESS', 'FAILED', 'PENDING') DEFAULT 'SUCCESS',
    payment_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id)
);
